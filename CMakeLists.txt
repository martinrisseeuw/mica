cmake_minimum_required(VERSION 3.21)
project(Mica LANGUAGES C CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "Mica currently targets macOS.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(PROJECT_ARCH "arm64" CACHE STRING "CEF architecture")
set(CEF_ROOT "${CMAKE_SOURCE_DIR}/vendor/cef" CACHE PATH "CEF binary distribution")
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
set(SPARKLE_ROOT "${CMAKE_SOURCE_DIR}/vendor/sparkle" CACHE PATH "Sparkle binary distribution")
find_library(SPARKLE_FRAMEWORK Sparkle PATHS "${SPARKLE_ROOT}" NO_DEFAULT_PATH REQUIRED)
add_executable(Mica src/main.mm src/Appearance.mm src/WindowChrome.mm src/Updates.mm src/DefaultBrowser.mm src/Browser.mm src/StartPage.mm src/ExtensionPanel.mm src/Model.mm src/History.mm src/Shield.mm src/NativeMessaging.mm src/Profile.mm test/UpdateProbe.mm)
add_executable(MicaHelper src/helper.cc)
foreach(target Mica MicaHelper)
  target_include_directories(${target} PRIVATE "${CEF_ROOT}" src)
  target_compile_definitions(${target} PRIVATE CEF_USE_SANDBOX)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
  target_link_libraries(${target} PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
endforeach()
target_compile_options(Mica PRIVATE -fobjc-arc)
target_link_libraries(Mica PRIVATE "-framework Security" "-framework QuartzCore" "${SPARKLE_FRAMEWORK}")
set_target_properties(Mica PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks")
add_executable(ModelTests test/model_tests.mm src/Model.mm src/History.mm src/Shield.mm src/NativeMessaging.mm src/Profile.mm)
target_include_directories(ModelTests PRIVATE src)
target_compile_options(ModelTests PRIVATE -fobjc-arc -Wall -Wextra -UNDEBUG)
target_link_libraries(ModelTests PRIVATE "-framework Foundation" "-framework Security")
enable_testing()
add_test(NAME ModelTests COMMAND ModelTests "${CMAKE_SOURCE_DIR}/resources/shield-rules.json")
